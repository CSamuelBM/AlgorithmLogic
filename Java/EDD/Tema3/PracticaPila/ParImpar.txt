import java.util.Scanner;
import java.util.Stack;

public class ParImpar{
   public static void main(String[]args){
      Scanner cs= new Scanner(System.in);
      Stack<Integer> pilaA= new Stack<>();
      Stack<Integer> pilaB= new Stack<>();
      
      int opc= 0;

      while(opc!= 7){
            System.out.println("1.- Registrar");
            System.out.println("2.- Mostrar");
            System.out.println("3.- Eliminar");
            System.out.println("4.- Ver Elemento");
            System.out.println("5.- Buscar Elemento");
            System.out.println("6.- Estado");
            System.out.println("7.- Salir");
              opc= cs.nextInt();
      switch(opc){
          case 1:
               System.out.println("Ingresa Un Numero");  
                 int num= cs.nextInt();

               if(num% 2== 0)
                  pilaA.push(num);
               else pilaB.push(num);
               
              System.out.println("Registro Exitoso!!"); 
            break;
 
          case 2:
             System.out.println("1.- Pila A");  
             System.out.println("2.- Pila B"); 
               opc= cs.nextInt();
             switch(opc){
                case 1:
                    if(!pilaA.empty())
                      System.out.println("Elementos De Pila A Son: "+ pilaA); 
                    else System.out.println("No Hay Elementos"); 
                  break;
                case 2:
                   if(!pilaB.empty())
                      System.out.println("Elementos De Pila B Son: "+ pilaB); 
                    else System.out.println("No Hay Elementos"); 
                  break;
              }  
           break;  

           case 3:
               System.out.println("1.- Pila A");  
               System.out.println("2.- Pila B"); 
                 opc= cs.nextInt();
             switch(opc){
                case 1:
                    if(!pilaA.empty()){
                       pilaA.pop();
                       System.out.println("Eliminado Exitoso!!"); 
                     }
                    else System.out.println("No Hay Elementos"); 
                  break;
                case 2:
                   if(!pilaB.empty()){
                       pilaB.pop();
                       System.out.println("Eliminado Exitoso!!"); 
                     }
                    else System.out.println("No Hay Elementos"); 
                  break;
              }  
             break; 

           case 4:
               System.out.println("1.- Pila A");  
               System.out.println("2.- Pila B"); 
                 opc= cs.nextInt();
             switch(opc){
                case 1:
                    if(!pilaA.empty()){
                       System.out.println("El Ultimo Elemento Ingresado De Pila A Es: "+ pilaA.peek()); 
                     }
                    else System.out.println("No Hay Elementos"); 
                  break;
                case 2:
                   if(!pilaB.empty()){
                       System.out.println("El Ultimo Elemento Ingresado De Pila B Es: "+ pilaB.peek());  
                     }
                    else System.out.println("No Hay Elementos"); 
                  break;
              }  
             break;     

           case 5:
                 System.out.println("1.- Pila A");  
                 System.out.println("2.- Pila B"); 
                   opc= cs.nextInt();
             switch(opc){
                case 1:
                    if(!pilaA.empty()){
                       System.out.println("Elemento A Buscar");
                       int numBuscar= cs.nextInt();

                       if(pilaA.search(numBuscar)!= -1)
                           System.out.println("El Elemento Se Encontro El Pila A");
                       else System.out.println("No Se Encontro El Elemento En Pila A");
                     }
                    else System.out.println("No Hay Elementos"); 
                  break;
                case 2:
                   if(!pilaB.empty()){
                       System.out.println("Elemento A Buscar");
                       int numBuscar= cs.nextInt();

                       if(pilaB.search(numBuscar)!= -1)
                           System.out.println("El Elemento Se Encontro El Pila B");
                       else System.out.println("No Se Encontro El Elemento En Pila B");  
                     }
                    else System.out.println("No Hay Elementos"); 
                  break;
              }  
            break; 

           case 6:
               System.out.println("1.- Pila A");  
               System.out.println("2.- Pila B"); 
                 opc= cs.nextInt();
             switch(opc){
                case 1:
                    if(!pilaA.empty()){
                       System.out.println("Hay Elementos En Pila A"); 
                     }
                    else System.out.println("No Hay Elementos En Pila A"); 
                  break;
                case 2:
                   if(!pilaB.empty()){
                       System.out.println("Hay Elementos En Pila B"); 
                     }
                    else System.out.println("No Hay Elementos En Pila B"); 
                  break;
              }  
             break;   

           case 7: 
              System.out.println("Saliste-->");
           return;
           default: System.out.println("Opcion No Valida");

      }//fin switch
    }//fin while
  } //fin main
} //fin class
